DROP TABLE AUTHORS cascade constraints;
DROP TABLE BOOK_ITEMS cascade constraints;
DROP TABLE BOOKS cascade constraints;
DROP TABLE BORROWING cascade constraints;
DROP TABLE CUSTOMERS cascade constraints;
DROP TABLE WRITING cascade constraints;

  CREATE TABLE AUTHORS 
   (	AUTHOR_ID NUMBER(10,0), 
	LAST_NAME VARCHAR2(30 CHAR), 
	FIRST_NAME VARCHAR2(20 CHAR), 
	BIRTH_DATE DATE
   ) ;

  CREATE TABLE BOOK_ITEMS 
   (	BOOK_ITEM_ID VARCHAR2(20 CHAR), 
	BOOK_ID NUMBER(10,0), 
	THEORETICAL_VALUE NUMBER(10,2)
   ) ;

  CREATE TABLE BOOKS 
   (	BOOK_ID NUMBER(10,0), 
	TITLE VARCHAR2(200 CHAR), 
	ISBN VARCHAR2(13 CHAR), 
	PUBLISHER VARCHAR2(200 CHAR), 
	PUBLISHING_DATE DATE, 
	PRICE NUMBER(8,0), 
	TOPIC VARCHAR2(30 CHAR), 
	NUMBER_OF_PAGES NUMBER(4,0)
   ) ;

  CREATE TABLE BORROWING 
   (	CUSTOMER_ID VARCHAR2(20 CHAR), 
	BOOK_ITEM_ID VARCHAR2(20 CHAR), 
	BORROWING_DATE DATE, 
	BORROWING_PERIOD_IN_DAYS NUMBER(3,0) DEFAULT 30, 
	BRING_BACK_DATE DATE, 
	LATE_FEE NUMBER(6,0)
   ) ;

  CREATE TABLE CUSTOMERS 
   (	LIBRARY_CARD_NUMBER VARCHAR2(20 CHAR), 
	LAST_NAME VARCHAR2(30 CHAR), 
	FIRST_NAME VARCHAR2(20 CHAR), 
	BIRTH_DATE DATE, 
	GENDER CHAR(1 CHAR), 
	ADDRESS VARCHAR2(200 CHAR), 
	ENROLLMENT_DATE DATE, 
	MEMBERSHIP_FEE NUMBER(5,0), 
	CATEGORY VARCHAR2(20 CHAR)
   ) ;

  CREATE TABLE WRITING 
   (	AUTHOR_ID NUMBER(10,0), 
	BOOK_ID NUMBER(10,0), 
	ROYALTY NUMBER
   ) ;
