DROP TABLE O_COUNTRIES purge;

CREATE TABLE O_COUNTRIES
   (	C_ID NUMBER(5), 
	NAME VARCHAR2(50), 
	AREA NUMBER(10), 
	POPULATION NUMBER(10), 
	CAPITAL VARCHAR2(40),
        CONTINENT VARCHAR2(50), 
	 CONSTRAINT O_C_PK PRIMARY KEY (C_ID),
         CONSTRAINT O_C_UK_NAME UNIQUE (NAME));
  
CREATE TABLE O_MEDAL_TABLE
   (	COUNTRY_ID NUMBER(5), 
	GOLD NUMBER(2), 
	SILVER NUMBER(2), 
	BRONZE NUMBER(2), 
	 CONSTRAINT O_MT_PK PRIMARY KEY (COUNTRY_ID),
  	 CONSTRAINT O_MT_FK FOREIGN KEY (COUNTRY_ID)
	      REFERENCES O_COUNTRIES (C_ID));
 
CREATE TABLE O_ATHLETES
    (  A_ID NUMBER(5),
       NAME VARCHAR2(50),
       BIRTHDATE DATE,
       COUNTRY_ID NUMBER(5),
       TEAM VARCHAR2(3),
       BIRTHPLACE VARCHAR2(50),
        CONSTRAINT O_A_PK PRIMARY KEY(A_ID),
        CONSTRAINT O_A_UK UNIQUE (NAME,COUNTRY_ID,BIRTHDATE),
        CONSTRAINT O_A_FK_C FOREIGN KEY (COUNTRY_ID)
             REFERENCES O_COUNTRIES (C_ID) );
          
  CREATE TABLE O_TEAMMEMBERS
   (	ATHLETE_ID NUMBER(5), 
	TEAM_ID NUMBER(5), 
	 CONSTRAINT O_TM_PK PRIMARY KEY (ATHLETE_ID, TEAM_ID), 
	 CONSTRAINT O_TM_FK_T FOREIGN KEY (TEAM_ID)
	          REFERENCES O_ATHLETES (A_ID), 
	 CONSTRAINT O_TM_FK_M FOREIGN KEY (ATHLETE_ID)
	          REFERENCES O_ATHLETES (A_ID));
          
  CREATE TABLE O_SPORT_DISCIPLINES
   (	D_ID NUMBER(5), 
	NAME VARCHAR2(50), 
	 CONSTRAINT O_D_PK PRIMARY KEY (D_ID),
  	 CONSTRAINT O_D_UK UNIQUE (NAME));
        
   CREATE TABLE O_SPORT_EVENTS
   (	E_ID NUMBER(5), 
	DISCIPLINE_ID NUMBER(5), 
	NAME VARCHAR2(50), 
	GENDER VARCHAR2(5), 
	 CONSTRAINT O_E_PK PRIMARY KEY (E_ID),
  	 CONSTRAINT O_E_UK UNIQUE (DISCIPLINE_ID, NAME, GENDER),
  	 CONSTRAINT O_E_FK_D FOREIGN KEY (DISCIPLINE_ID)
	  REFERENCES O_SPORT_DISCIPLINES (D_ID));

CREATE TABLE O_RESULTS
   (   ATHLETE_ID NUMBER(5),
       EVENT_ID NUMBER(5),
       PLACE NUMBER(3),
       NOTE VARCHAR2(150),
        CONSTRAINT O_R_PK PRIMARY KEY  (ATHLETE_ID, EVENT_ID),
        CONSTRAINT O_R_FK_E FOREIGN KEY ( EVENT_ID)
                  REFERENCES O_SPORT_EVENTS (E_ID),
        CONSTRAINT O_R_FK_A FOREIGN KEY (ATHLETE_ID)
                  REFERENCES O_ATHLETES (A_ID) );
